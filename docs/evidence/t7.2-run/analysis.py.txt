# T7.2 scored run - the report's analysis (run registration, the analysis as amended by Addendum 1).
#
#   cd docs/evidence/t7.2-run
#   python3 analysis.py.txt scored scored/results . > scored/analysis-output.txt
#
# RECORDS  : the run's record archive, unpacked (done.tsv, sN-record/ for every Faultline slot,
#            and claudecode-usage.tsv)
# RESULTS  : SREGym's per-attempt results files, in SREGym's own layout
#            (<run>/<arm>/<problem>/run_1/<problem>_results.csv)
# EVIDENCE : docs/evidence/t7.2-run/ (scaled-sample.tsv)
#
# Claude Code's session logs are 19 MB and are kept off the repository. Their token counts were
# extracted once, from collect's archive, with
#   python3 analysis.py.txt usage <SREGym's results/ directory> > scored/claudecode-usage.tsv
# and the analysis reads that file. The extraction is the tally stage's own rule: assistant
# messages, each message id counted once.
#
# Everything printed is computed from those files, except TRUTH below, which is read by hand
# from the judge's own records and printed beside its source so that it can be checked.
# Standard library only. The bootstrap seed is fixed, so the output is reproducible.

def judged(arm, problem):
    """The attempt's SREGym record: the run_1 results file whose run completed (a re-run slot has
    two directories, the first incomplete)."""
    found = []
    for path in sorted(glob.glob(f"{RES}/*/{arm}/{problem}/run_1/{problem}_results.csv")):
        r = next(csv.DictReader(open(path)))
        found.append((path, r))
    complete = [(p, r) for p, r in found if r.get("run_status") == "complete"]
    return complete[-1] if complete else (found[-1] if found else (None, None)), found

# --- helpers ----------------------------------------------------------------------------------


def pct(x):
    return f"{100 * x:.0f} %"


def boot(values, rng):
    """Percentile bootstrap over problems: the mean of a resample of `values`, BOOT times."""
    n = len(values)
    means = sorted(sum(values[rng.randrange(n)] for _ in range(n)) / n for _ in range(BOOT))
    return means[int(0.025 * BOOT)], means[int(0.975 * BOOT) - 1]


def wilson(k, n, z=1.959964):
    p = k / n
    c = (p + z * z / (2 * n)) / (1 + z * z / n)
    h = z * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n)) / (1 + z * z / n)
    return max(0.0, c - h), min(1.0, c + h)


NOT_ESTABLISHED = re.compile(r"^Root cause: (Not established|The (evidence|board) does not establish)", re.M)


def category(problem, arm):
    """The registered five categories, for a failed attempt. The rule, fixed before it is applied:
    4 if the attempt submitted no diagnosis: `Faultline reached no verdict.` (triage's gate), a
      verdict whose root cause says it is not established, or no submission (an agent timeout);
    5 if the run did not complete (none did, after the registered re-run);
    else by the judge's dimensions: D1 < 0.67 localisation, D2 < 0.67 characterisation, else scope."""
    a = A[(problem, arm)]
    if a["status"] != "complete":
        return "5 adapter or harness failure"
    sub = a["submission"]
    if sub.startswith("Faultline reached no verdict") or NOT_ESTABLISHED.search(sub) or not sub.strip():
        return "4 no verdict or timeout"
    if a["D"].get("D1", 0) < 0.67:
        return "1 localisation miss"
    if a["D"].get("D2", 0) < 0.67:
        return "2 characterisation miss"
    return "3 scope miss"


def fl_cost(problem):
    f = F[problem]
    return f["tin"] * 5e-6 + f["tout"] * 25e-6 if (f["tin"] or f["tout"]) else 0.05


USAGE = list(csv.DictReader(open(f"{REC}/claudecode-usage.tsv"), delimiter="\t"))


def cc_cost(problem):
    """Sonnet 4.6: $3 input, $15 output, $0.30 cache read, $3.75 cache write, per million."""
    return sum(int(u["input"]) * 3e-6 + int(u["output"]) * 15e-6 + int(u["cache_read"]) * 0.3e-6
               + int(u["cache_write"]) * 3.75e-6 for u in USAGE if u["problem"] == problem)


def minutes(problem, arm):
    row = SLOT[(problem, arm)]
    from datetime import datetime
    fmt = "%Y-%m-%d %H:%M:%S UTC"
    return (datetime.strptime(row[7], fmt) - datetime.strptime(row[6], fmt)).total_seconds() / 60
