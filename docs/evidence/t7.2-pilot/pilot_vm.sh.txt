#!/usr/bin/env bash
# T7.2 dev read and dev pilot, on the deployment. Registered in
# evals/runs/PREREGISTRATION-T7.2-pilot.md before any stage is run.
#
# Copied to the VM as /tmp/pilot_vm.sh, with these beside it in /tmp:
#   faultline-agent.patch      SREGym's side (evals/sregym/), applied at the pin
#   faultline-src.tar.gz       the box's source bundle (part B only)
#   faultline-seed.tar.gz      the seeding container's bundle (part B only)
#   kind_measure_1c.py         1c's sampler, unchanged (docs/evidence/t7.2-kind-host/)
# Run one stage at a time from the Mac, each output saved to ~/Downloads/t72-pilot-<stage>.txt and
# read before the next:
#     ssh "$VM" 'bash /tmp/pilot_vm.sh <stage> [args]'     (host-on/host-off: ssh -t ... sudo bash)
#
# Part A, the dev read ($0, no key on the VM):
#   preread, killswitch-on, host-on, install, cluster, devread shop, podstate shop (Addendum 1),
#   cluster-reset, devread hotel, cluster-reset, devread social, teardown, host-off, cleanup,
#   killswitch-off, incidents
# Part B, the dev pilot (after the dev read's profile patch is merged):
#   preread, killswitch-on, host-on, install, cluster, bench-db (bench-reset to redo it,
#   Addendum 4), serve-on, key-check,
#   judge-check, hold (nohup, the sampler), attempt shop faultline (nohup) with loki-watch shop
#   beside it (nohup, Addendum 5: loki-read while Loki exists), attempt shop claudecode (nohup)
#   with probe shop beside it (nohup, Addendum 6), the other four attempts, collect, teardown,
#   host-off, cleanup, killswitch-off, incidents
#
# The directory is 1c's (~/t7.2-1c, its kubeconfig at ~/t7.2-1c/kubeconfig and copied to
# ~/.kube/config), so 1c's sampler runs unchanged, as in the topology runs.

set -uo pipefail
D="$HOME/t7.2-1c"
BIN="$D/bin"
SR="$D/SREGym"
SREGYM_PIN=46c853db
PG=t72-bench-pg
GW=172.17.0.1
stage="${1:?stage}"
ts() { date -u '+%Y-%m-%d %H:%M:%S UTC'; }
kenv() { export PATH="$BIN:$HOME/.local/bin:$PATH" KUBECONFIG="$D/kubeconfig"; }
renv() {
    # SREGym's environment for a run: KUBECONFIG unset (it reads ~/.kube/config by fixed path),
    # Helm's homes under the run's directory, its API on every interface - the boxed agents reach
    # it through the egress proxy from docker0, and host-on's rules admit docker0 alone.
    export PATH="$BIN:$HOME/.local/bin:$D/uvbin:$PATH"
    unset KUBECONFIG
    export HELM_CONFIG_HOME="$D/helm/config" HELM_CACHE_HOME="$D/helm/cache" HELM_DATA_HOME="$D/helm/data"
    export UV_CACHE_DIR="$D/uv-cache"
}

app_vars() {
    case "$1" in
        shop)   PROBLEM=edge_request_filter_cpu_saturation; NS=astronomy-shop ;;
        hotel)  PROBLEM=update_incompatible_correlated;     NS=hotel-reservation ;;
        social) PROBLEM=k8s_target_port-misconfig;          NS=social-network ;;
        *) echo "unknown app: $1"; exit 2 ;;
    esac
}

world() {
    echo "== memory, load, disk"
    free -m | sed 's/^/  /'; echo "  load: $(cat /proc/loadavg)"; df -h / | sed 's/^/  /'
    echo "== containers running: $(docker ps -q | wc -l); kind nodes: $(docker ps -aq --filter label=io.x-k8s.kind.cluster | wc -l)"
    echo "== firing alerts on the world (Alertmanager :9093)"
    curl -fsS -m 10 'http://localhost:9093/api/v2/alerts?active=true&silenced=false&inhibited=false' \
        | python3 -c 'import json,sys; a=[x["labels"].get("alertname","?")+"/"+x["labels"].get("service_name","?") for x in json.load(sys.stdin)]; print("  %d: %s" % (len(a), ", ".join(a) or "none"))' \
        || echo "  alertmanager unreadable"
    echo "== git status of ~/faultline (empty means committed configuration)"
    (cd ~/faultline && git status --short | sed 's/^/  /'); echo "  (end)"
}

prom() {
    # One instant query through the API server's service proxy, read-only.
    local q; q=$(python3 -c 'import sys, urllib.parse; print(urllib.parse.quote(sys.argv[1]))' "$1")
    kubectl get --raw "/api/v1/namespaces/observe/services/prometheus-server:80/proxy/api/v1/query?query=$q" \
        | python3 -c 'import json,sys; r=json.load(sys.stdin)["data"]["result"]; print("  %d series" % len(r)); [print("   ", e["metric"], e.get("value",[None,None])[1]) for e in r[:40]]' \
        || echo "  (query failed)"
}

case "$stage" in

preread)
    echo "pilot pre-read at $(ts) on $(hostname), kernel $(uname -r), $(uptime -p)"
    echo "== reboot pending: $([ -e /var/run/reboot-required ] && echo YES || echo no)"
    echo "== timezone: $(timedatectl show -p Timezone --value 2>/dev/null || echo unknown); date +%Z: $(date +%Z)"
    echo "== ports: $(for p in 8000 9954 16443 55432 55480; do printf '%s %s; ' $p "$(ss -ltnH "sport = :$p" | grep -q . && echo IN-USE || echo free)"; done)"
    echo "== $D: $([ -e "$D" ] && echo present || echo absent); key file: $([ -e "$D/key" ] && echo PRESENT || echo absent)"
    echo "== files in /tmp: $(ls /tmp/pilot_vm.sh /tmp/faultline-agent.patch /tmp/faultline-src.tar.gz /tmp/faultline-seed.tar.gz /tmp/kind_measure_1c.py 2>&1 | tr '\n' ' ')"
    echo "== the kill switch now"
    (cd ~/faultline/deploy && grep -n "KILL_SWITCH" compose.yml | sed 's/^/  /')
    world
    ;;

killswitch-on|killswitch-off)
    cd ~/faultline/deploy || exit 1
    if [ "$stage" = killswitch-on ]; then
        sed -i 's/FAULTLINE_EXECUTOR_KILL_SWITCH: "0"/FAULTLINE_EXECUTOR_KILL_SWITCH: "1"/' compose.yml
    else
        sed -i 's/FAULTLINE_EXECUTOR_KILL_SWITCH: "1"/FAULTLINE_EXECUTOR_KILL_SWITCH: "0"/' compose.yml
    fi
    grep -n "KILL_SWITCH" compose.yml
    docker compose up -d executor
    for _ in $(seq 1 10); do
        docker compose exec -T executor curl -fsS localhost:8100/healthz < /dev/null && break
        sleep 3
    done
    echo; echo "git status:"; cd ~/faultline && git status --short; echo "(git status end)"
    ;;

host-on)
    # Run with sudo. SREGym's API (8000), MCP forward (9954) and Kubernetes proxy (16443) are
    # dropped from every interface but the loopback, as in every T7.2 run so far; then docker0 is
    # admitted for those three and for the benchmark database (55432) and the bundle (55480),
    # because the boxed agents reach the host through the egress proxy on the default bridge
    # (stage 0, Addendum 1). The ACCEPTs are inserted last, so they sit above the DROPs.
    # Addendum 3: refuse a second application, which would leave a set host-off does not remove.
    if iptables -C INPUT -i docker0 -p tcp --dport 55480 -j ACCEPT 2>/dev/null; then
        echo "REFUSED: host-on's rules are already in place"; iptables -S INPUT | grep -E 'dport (8000|9954|16443|55432|55480)'; exit 1
    fi
    for p in 8000 9954 16443; do iptables -I INPUT ! -i lo -p tcp --dport "$p" -j DROP; done
    for p in 8000 9954 16443 55432 55480; do iptables -I INPUT -i docker0 -p tcp --dport "$p" -j ACCEPT; done
    iptables -S INPUT | grep -E 'dport (8000|9954|16443|55432|55480)'
    sysctl -w fs.inotify.max_user_instances=1024 fs.inotify.max_user_watches=1048576
    ;;

host-off)
    # Run with sudo.
    # Addendum 3: every copy, so a doubled host-on is undone too.
    for p in 8000 9954 16443 55432 55480; do
        while iptables -C INPUT -i docker0 -p tcp --dport "$p" -j ACCEPT 2>/dev/null; do iptables -D INPUT -i docker0 -p tcp --dport "$p" -j ACCEPT; done
    done
    for p in 8000 9954 16443; do
        while iptables -C INPUT ! -i lo -p tcp --dport "$p" -j DROP 2>/dev/null; do iptables -D INPUT ! -i lo -p tcp --dport "$p" -j DROP; done
    done
    iptables -S INPUT | grep -E 'dport (8000|9954|16443|55432|55480)' || echo "rules gone"
    sysctl -w fs.inotify.max_user_instances=128 fs.inotify.max_user_watches=124032
    ;;

install)
    set -e
    echo "install at $(ts)"
    [ -e "$SR" ] && { echo "REFUSED: $SR exists; tear down and clean up first (Addendum 3)"; exit 1; }
    mkdir -p "$BIN" "$D/uvbin" "$D/out" && cd "$BIN"
    curl -fsSLo kind https://kind.sigs.k8s.io/dl/v0.27.0/kind-linux-amd64
    echo "$(curl -fsSL https://kind.sigs.k8s.io/dl/v0.27.0/kind-linux-amd64.sha256sum | awk '{print $1}')  kind" | sha256sum -c
    curl -fsSLo kubectl https://dl.k8s.io/release/v1.32.1/bin/linux/amd64/kubectl
    echo "$(curl -fsSL https://dl.k8s.io/release/v1.32.1/bin/linux/amd64/kubectl.sha256)  kubectl" | sha256sum -c
    curl -fsSLo helm.tgz https://get.helm.sh/helm-v4.3.0-linux-amd64.tar.gz
    echo "$(curl -fsSL https://get.helm.sh/helm-v4.3.0-linux-amd64.tar.gz.sha256sum | awk '{print $1}')  helm.tgz" | sha256sum -c
    tar -xzf helm.tgz --strip-components=1 linux-amd64/helm && rm helm.tgz
    chmod +x kind kubectl helm
    command -v uv >/dev/null || curl -fsSL https://astral.sh/uv/install.sh | env UV_INSTALL_DIR="$D/uvbin" UV_NO_MODIFY_PATH=1 sh
    renv
    echo "uv: $(uv --version)"
    git clone -q https://github.com/SREGym/SREGym.git "$SR"
    cd "$SR"
    git checkout -q "$SREGYM_PIN" && git log --oneline -1
    git submodule update --init SREGym-applications
    (cd SREGym-applications && git submodule update --init astronomy-shop)
    git apply /tmp/faultline-agent.patch && echo "faultline-agent.patch applied"
    git status --short | sed 's/^/  /'
    uv sync --frozen 2>&1 | tail -2
    # Part B only, and recorded: claudecode's version, pinned to npm's latest now (registration).
    if [ -e /tmp/faultline-src.tar.gz ]; then
        v=$(curl -fsS https://registry.npmjs.org/@anthropic-ai/claude-code/latest | python3 -c 'import json,sys; print(json.load(sys.stdin)["version"])')
        python3 - "$v" <<'EOF'
import re, sys
p = "agents.yaml"; s = open(p).read()
block = re.search(r"(  - name: claudecode\n(?:    .*\n)+?)    agent_version: null\n", s)
assert block, "claudecode row not found"
s = s.replace(block.group(0), block.group(1) + f'    agent_version: "{sys.argv[1]}"\n', 1)
open(p, "w").write(s)
print("claudecode pinned to", sys.argv[1])
EOF
        sha256sum /tmp/faultline-src.tar.gz /tmp/faultline-seed.tar.gz
    fi
    ;;

cluster)
    kenv
    cd "$SR" || exit 1
    date -u
    bash kind/setup_kind_cluster.sh
    kubectl get nodes
    mkdir -p "$HOME/.kube" && install -m 600 "$D/kubeconfig" "$HOME/.kube/config"
    cmp -s "$D/kubeconfig" "$HOME/.kube/config" && echo "~/.kube/config: identical" || echo "~/.kube/config: DIFFERS"
    ;;

cluster-reset)
    # Between dev problems: a fresh cluster, so each application deploys as it would first.
    kenv
    echo "cluster-reset at $(ts)"
    pkill -f '^kubectl port-forward svc/' || true
    kind delete cluster
    docker network rm kind
    bash /tmp/pilot_vm.sh cluster
    ;;

devread)
    # Part A. SREGym's external-harness deploy (1c's, no agent, no key), then read-only reads.
    app_vars "${2:?app}"
    renv
    export API_BIND_HOST=127.0.0.1
    echo "key-like variables in env: $(env | grep -cE '^[A-Za-z_]*(KEY|TOKEN|SECRET)[A-Za-z_]*=')"
    cd "$SR" || exit 1
    echo "deploy $PROBLEM at $(ts)"
    timeout 45m uv run python main.py --use-external-harness --problem "$PROBLEM" --agent demo 2>&1 | tail -5
    echo "deploy exit ${PIPESTATUS[0]} at $(ts)"
    kenv
    sleep 120
    echo "== reads at $(ts), namespace $NS"
    echo "== workloads (kind, name, containers)"
    kubectl -n "$NS" get deploy,sts -o custom-columns=KIND:.kind,NAME:.metadata.name,CONTAINERS:.spec.template.spec.containers[*].name | sed 's/^/  /'
    echo "== Jaeger services"
    kubectl get --raw "/api/v1/namespaces/observe/services/jaeger-out:16686/proxy/api/services" | sed 's/^/  /'; echo
    echo "== span-metric families";             prom 'count by (__name__) ({__name__=~"traces_span_metrics.*"})'
    echo "== span-metric services";             prom 'count by (service_name) (traces_span_metrics_calls_total)'
    echo "== kube_pod_owner in $NS";            prom "count(kube_pod_owner{namespace=\"$NS\"})"
    echo "== kube_replicaset_owner in $NS";     prom "count(kube_replicaset_owner{namespace=\"$NS\"})"
    echo "== cAdvisor working set in $NS";      prom "count(container_memory_working_set_bytes{namespace=\"$NS\",container!=\"\",container!=\"POD\"})"
    echo "== the alarms, evaluated once now (registered rules, read-only)"
    prom "max by (pod) (max_over_time(kube_pod_container_status_waiting_reason{namespace=\"$NS\",reason=\"CrashLoopBackOff\"}[5m])) >= 1"
    prom "max by (pod) (kube_pod_status_phase{namespace=\"$NS\",phase=~\"Pending|Unknown|Failed\"}) > 0"
    prom "kube_deployment_spec_replicas{namespace=\"$NS\"} != kube_deployment_status_replicas_available{namespace=\"$NS\"}"
    echo "== pods in $NS, with restarts (Addendum 1)"
    kubectl -n "$NS" get pods --no-headers | sed 's/^/  /'
    echo "== Loki: not read here. SREGym skips Loki under the external harness (main.py:"
    echo "   deploy_loki=not args.use_external_harness); part B's loki-read reads it (Addendum 1)"
    echo "== timezone: $(timedatectl show -p Timezone --value 2>/dev/null); date +%Z: $(date +%Z)"
    ;;

podstate)
    # Addendum 1. Read-only: the pods of a deployed dev problem, and every container that has
    # restarted, with its last termination. For the shop, run once before its cluster-reset.
    app_vars "${2:?app}"
    kenv
    echo "pod state in $NS at $(ts)"
    kubectl -n "$NS" get pods --no-headers | sed 's/^/  /'
    kubectl -n "$NS" get pods -o json | python3 -c '
import json, sys
for pod in json.load(sys.stdin)["items"]:
    for c in pod.get("status", {}).get("containerStatuses", []) or []:
        if c.get("restartCount"):
            last = (c.get("lastState") or {}).get("terminated") or {}
            print("  %s/%s restarts=%s last=%s exit=%s at %s..%s" % (pod["metadata"]["name"], c["name"],
                c["restartCount"], last.get("reason"), last.get("exitCode"), last.get("startedAt"), last.get("finishedAt")))'
    echo "(end)"
    ;;

loki-read)
    # Addendum 1. Part B, read-only, after attempt 1 has deployed Loki: its service, its labels and
    # the pod values for the namespace - the labels McpToolSet's selector uses. If `namespace`
    # and `pod` are not both among them, the pilot stops before attempt 2.
    app_vars "${2:?app}"
    kenv
    echo "loki-read at $(ts)"
    kubectl get svc -A --no-headers | grep -iE 'loki|promtail' | sed 's/^/  /'
    lns=$(kubectl get svc -A --no-headers | awk '$2 == "loki" {print $1; exit}')
    echo "loki service namespace: ${lns:-NONE}"
    [ -n "$lns" ] || exit 1
    kubectl get --raw "/api/v1/namespaces/$lns/services/loki:3100/proxy/loki/api/v1/labels" | sed 's/^/  /'; echo
    q=$(python3 -c 'import sys, urllib.parse; print(urllib.parse.quote(sys.argv[1]))' "{namespace=\"$NS\"}")
    kubectl get --raw "/api/v1/namespaces/$lns/services/loki:3100/proxy/loki/api/v1/label/pod/values?query=$q" \
        | python3 -c 'import json,sys; v=json.load(sys.stdin).get("data") or []; print("  pod values for the namespace: %d, e.g. %s" % (len(v), v[:6]))'
    ;;

loki-watch)
    # Addendum 5. Part B, read-only, started beside attempt 1. SREGym's cleanup deletes every
    # namespace not in the bare cluster's baseline (cluster_state.reconcile_to_baseline), and
    # Loki's, `observe`, is one, so loki-read cannot run after the attempt. This polls every 30 s
    # and runs loki-read once, while Loki exists:
    #   - when Loki has pod values for the namespace, or
    #   - after ten polls (5 min) in which Loki answered with labels but no such values, so the
    #     label list is read either way.
    # If the attempt ends first, it says so. Addendum 1's rule is unchanged.
    app_vars "${2:?app}"
    f="${3:?the output file of the attempt}"
    kenv
    echo "loki-watch from $(ts), for $NS, until $f says exit"
    q=$(python3 -c 'import sys, urllib.parse; print(urllib.parse.quote(sys.argv[1]))' "{namespace=\"$NS\"}")
    answered=0
    for i in $(seq 1 160); do
        if grep -q '^exit ' "$f" 2>/dev/null; then
            echo "the attempt ended before a reading, at poll $i, $(ts)"; bash "$0" loki-read "$2"; exit 1
        fi
        lns=$(kubectl get svc -A --no-headers 2>/dev/null | awk '$2 == "loki" {print $1; exit}')
        if [ -n "$lns" ]; then
            base="/api/v1/namespaces/$lns/services/loki:3100/proxy/loki/api/v1"
            nl=$(kubectl get --raw "$base/labels" 2>/dev/null | python3 -c 'import json,sys; print(len(json.load(sys.stdin).get("data") or []))' 2>/dev/null)
            np=$(kubectl get --raw "$base/label/pod/values?query=$q" 2>/dev/null | python3 -c 'import json,sys; print(len(json.load(sys.stdin).get("data") or []))' 2>/dev/null)
            [ "${nl:-0}" -gt 0 ] && answered=$((answered + 1))
            if [ "${np:-0}" -gt 0 ] || [ "$answered" -ge 10 ]; then
                echo "reading at poll $i, $(ts): $nl labels, $np pod values for $NS"; bash "$0" loki-read "$2"; exit 0
            fi
        fi
        sleep 30
    done
    echo "no reading in 80 minutes"; exit 1
    ;;

probe)
    # Addendum 6. Part B, read-only, beside attempt 2 (Claude Code on the shop): the facts the
    # adapter's fixes need, measured on the live application while the fault is in. Waits until
    # Loki has the namespace's pods, then 300 s for the fault, then reads once. It prints
    # statuses, counts and sizes only: no log line, no object body, no value from a Secret.
    app_vars "${2:?app}"
    f="${3:?the output file of the attempt}"
    kenv
    echo "probe from $(ts), for $NS"
    q=$(python3 -c 'import sys, urllib.parse; print(urllib.parse.quote(sys.argv[1]))' "{namespace=\"$NS\"}")
    ready=0
    for i in $(seq 1 120); do
        if grep -q '^exit ' "$f" 2>/dev/null; then echo "the attempt ended before the probe, at poll $i"; exit 1; fi
        n=$(kubectl get --raw "/api/v1/namespaces/observe/services/loki:3100/proxy/loki/api/v1/label/pod/values?query=$q" 2>/dev/null | python3 -c 'import json,sys; print(len(json.load(sys.stdin).get("data") or []))' 2>/dev/null)
        if [ "${n:-0}" -gt 0 ]; then ready=$i; break; fi
        sleep 30
    done
    [ "$ready" -gt 0 ] || { echo "Loki never answered for $NS"; exit 1; }
    echo "Loki answered for $NS at poll $ready, $(ts); waiting 300 s for the fault"
    sleep 300
    grep -q '^exit ' "$f" 2>/dev/null && echo "(the attempt had ended by the time of the reading)"
    echo "reading at $(ts)"
    python3 - "$NS" <<'PY'
import json, subprocess, sys, time, urllib.parse

NS = sys.argv[1]
LOKI = "/api/v1/namespaces/observe/services/loki:3100/proxy/loki/api/v1/query_range"
PROM = "/api/v1/namespaces/observe/services/prometheus-server:80/proxy/api/v1/query"
CAP = 10000  # SREGym's kubectl server cuts every answer here (mcp_server/kubectl_server_helper/utils.py)
NOW = time.time()


def run(args):
    p = subprocess.run(args, capture_output=True, text=True, timeout=120)
    return p.returncode, p.stdout, " ".join(p.stderr.split())[:300]


def pods(name, shipped):
    # shipped: re.escape's spelling, as the adapter sends it today; corrected: the name as written.
    return (name.replace("-", "\\-") if shipped else name) + "-([a-z0-9]+-[a-z0-9]+|[0-9]+)"


print("== 1. Loki: the shipped selector and the corrected one, the last 30 min, limit 100")
for svc in ("product-catalog", "frontend-proxy"):
    for shipped in (True, False):
        sel = '{namespace="%s",pod=~"%s"}' % (NS, pods(svc, shipped))
        qs = urllib.parse.urlencode({"query": sel, "start": int((NOW - 1800) * 1e9), "end": int(NOW * 1e9), "limit": 100})
        rc, out, err = run(["kubectl", "get", "--raw", LOKI + "?" + qs])
        label = "shipped  " if shipped else "corrected"
        if rc:
            print(f"  {label} {sel}\n      REJECTED: {err}")
        else:
            res = json.loads(out)["data"]["result"]
            print(f"  {label} {sel}\n      accepted: {len(res)} streams, {sum(len(s['values']) for s in res)} lines")

print("\n== 2. Prometheus: span-call series by service and status, now")
rc, out, err = run(["kubectl", "get", "--raw", PROM + "?" + urllib.parse.urlencode(
    {"query": "count by (service_name, status_code) (traces_span_metrics_calls_total)"})])
if rc:
    print("  REJECTED:", err)
else:
    rows = sorted((r["metric"].get("service_name", "?"), r["metric"].get("status_code", "?"), r["value"][1])
                  for r in json.loads(out)["data"]["result"])
    for s, c, n in rows:
        print(f"  {s:20} {c:22} {n}")
    print("  services with an error-status series:", sorted({s for s, c, _ in rows if c == "STATUS_CODE_ERROR"}) or "none")

print("\n== 3. the adapter's range reads, as it sends them: (q)[1800s:15s] @ now")
ER = ('sum by(service_name) (rate(traces_span_metrics_calls_total{service_name="%s",status_code="STATUS_CODE_ERROR"}[5m])) '
      '/ sum by(service_name) (rate(traces_span_metrics_calls_total{service_name="%s"}[5m]))')
CR = 'sum by(service_name) (rate(traces_span_metrics_calls_total{service_name="%s"}[5m]))'
LP = ('histogram_quantile(0.95, sum by(service_name, le) (rate(traces_span_metrics_duration_milliseconds_bucket'
      '{service_name="%s",span_kind!="SPAN_KIND_INTERNAL"}[5m])))')
MEM = 'sum(container_memory_working_set_bytes{namespace="%s",pod=~"%s",container!="",container!="POD"})'
reads = []
for svc in ("frontend-proxy", "frontend", "product-catalog"):
    reads += [(f"error-ratio {svc}", ER % (svc, svc)), (f"call-rate {svc}", CR % svc), (f"latency-p95 {svc}", LP % svc)]
reads += [("memory frontend-proxy, shipped", MEM % (NS, pods("frontend-proxy", True))),
          ("memory frontend-proxy, corrected", MEM % (NS, pods("frontend-proxy", False)))]
for label, query in reads:
    sub = f"({query})[1800s:15s] @ {NOW:.3f}"
    rc, out, err = run(["kubectl", "get", "--raw", PROM + "?" + urllib.parse.urlencode({"query": sub})])
    if rc:
        print(f"  {label:34} REJECTED: {err}")
        continue
    res = json.loads(out)["data"]["result"]
    values = [float(v) for s in res for _, v in s.get("values", [])]
